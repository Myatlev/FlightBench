# RS7 session profile for the nexus-xr QEMU machine.
# Appended to boards/nexus-xr/NEXUS_XR.config in the custom-defaults blob (core/qemu/run.sh
# PROFILE=...), so the unmodified firmware boots configured: the CLI runs both on first boot.
#
# Sources: the owner's earlier SITL bench profile (servos re-centred to 1500 us, swashplate
# reverse flags removed - the FDM has no linkage geometry), plus the real board's output,
# governor and arming lines (diff all 2026-09-14,
# ~/rc-logs/rs7-fork/13092026/). Left out: FAKE baro/mag, GPS and nav modes (SITL-only),
# ESC_SENSOR/FREQ_SENSOR (head speed comes from bidirectional DShot here), OSD.

# --- outputs: motor 1 DShot600 with bidirectional telemetry, as on the board ---
feature GOVERNOR
feature TELEMETRY
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set dshot_burst = OFF
set use_unsynced_pwm = OFF
set motor_poles = 10,0,0,0

# --- servos 1-3 swashplate (120 deg), 4 tail; mixer ---
# Centres/travel and the swashplate flags as on that bench (all three swash servos without
# reverse: the FDM's swashplate has no mirrored linkage). Two signs differ from it
# and are the real board's again: tail servo +REV and mixer SP -740. Why: the SITL bench fed the
# FDM's FRD body rates straight into the firmware (pitch and yaw rate with the wrong sign for the
# firmware's +Z-up board frame, and the accelerometer upside down), and its profile compensated
# with those two flips. Here the IMU gets the physically right board frame (checked: the
# firmware's heading follows the FDM's), and the same two loops are then positive feedback with
# the SITL signs - measured 2026-09-25: tail pinned at its limit and ~1000 deg/s of yaw; a roll
# doublet flipped the model over pitch. With +REV / -740 both axes damp a doublet in ~0.6 s.
servo 1 1500 -500 500 1000 1000 560 0 2
servo 2 1500 -500 500 1000 1000 560 0 3
servo 3 1500 -500 500 1000 1000 560 0 2
servo 4 1500 -350 270 1000 1000 560 0 3
servo flags 1 -REV +GEO
servo flags 2 -REV +GEO
servo flags 3 -REV +GEO
servo flags 4 +REV +GEO
mixer input SR -1250 1250 740
mixer input SP -1250 1250 -740
mixer input SY -1250 1250 500
mixer input SC -1250 1250 1050

# --- transmission, battery, governor ---
set main_rotor_gear_ratio = 13,116
set tail_rotor_gear_ratio = 20,100
set battery_cell_count = 12
set tail_center_trim = 137
set gov_mode = ELECTRIC
set gov_throttle_type = SWITCH
set gov_autorotation_timeout = 60

# --- arming: ARM on AUX1 (channel 5 with map AECR1T23), as on the board ---
aux 0 0 0 1700 2100 0 0
# accelerometer calibrated: the "completed" key is crc16(UID) % 65521 + 2 of this
# machine's UID (nexus_xr.c), offsets 0 for an ideal chip
set acc_calibration = 0,0,0,-30671

# --- observation only: MSP_DEBUG carries requested/target/current head speed + PID terms ---
set debug_mode = GOVERNOR
